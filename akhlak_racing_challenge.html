Konfigurasi Web App URL Google Apps Script

Masukkan URL Apps Script Web App anda untuk menghubungkan keputusan permainan secara automatik ke Google Sheets.

Panduan Penyediaan Google Sheets & Apps Script:

  1. Buka Google Sheets baharu dan namakan: AKHLAK RACING CHALLENGE - REKOD MURID
  2. Buat 3 Helaian (Sheets) bernama: REKOD MURID, ANALISIS JAWAPAN, dan ANALISIS KELAS.
  3. Pergi ke menu Extensions > Apps Script.
  4. Padam semua kod sedia ada dan tampal kod Google Apps Script yang disediakan di bawah:
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // SHEET 1: REKOD MURID
    var sheet1 = ss.getSheetByName("REKOD MURID") || ss.insertSheet("REKOD MURID");
    if (sheet1.getLastRow() === 0) {
      sheet1.appendRow(["ID Sesi", "Tarikh", "Masa", "Nama Penuh", "Kelas", "Jumlah Soalan", "Jawapan Betul", "Jawapan Salah", "Jumlah Markah", "Peratus", "Masa Keseluruhan", "Status Permainan"]);
    }
    sheet1.appendRow([data.sessionId, data.date, data.time, data.studentName, data.studentClass, data.totalQuestions, data.correctCount, data.wrongCount, data.totalScore, data.percentage + "%", data.totalTimeSeconds + "s", data.status]);

    // SHEET 2: ANALISIS JAWAPAN
    var sheet2 = ss.getSheetByName("ANALISIS JAWAPAN") || ss.insertSheet("ANALISIS JAWAPAN");
    if (sheet2.getLastRow() === 0) {
      sheet2.appendRow(["ID Sesi", "Nama Murid", "Kelas", "Nombor Soalan", "Soalan", "Jawapan Murid", "Jawapan Betul", "Status Jawapan", "Markah", "Masa Menjawab"]);
    }
    if(data.answersDetail && data.answersDetail.length > 0) {
      data.answersDetail.forEach(function(item) {
        sheet2.appendRow([data.sessionId, data.studentName, data.studentClass, item.qNum, item.question, item.userAnswer, item.correctAnswer, item.isCorrect ? "BETUL" : "SALAH", item.score, item.timeTaken + "s"]);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("REKOD MURID");
  if(!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    result.push({
      sessionId: rows[i][0], date: rows[i][1], time: rows[i][2], studentName: rows[i][3], studentClass: rows[i][4],
      totalQuestions: rows[i][5], correctCount: rows[i][6], wrongCount: rows[i][7], totalScore: rows[i][8],
      percentage: rows[i][9], totalTimeSeconds: rows[i][10], status: rows[i][11]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
